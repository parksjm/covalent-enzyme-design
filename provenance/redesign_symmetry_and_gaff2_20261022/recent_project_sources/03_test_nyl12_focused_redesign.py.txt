def pdb_atom(
    serial: int,
    name: str,
    residue: str,
    chain: str,
    number: int,
    xyz: tuple[float, float, float],
    element: str = "C",
) -> str:
    """Construct a fixed-column test PDB atom."""
    record = "HETATM" if chain == "I" else "ATOM  "
    return (
        f"{record}{serial:5d} {name:>4s} {residue:3s} {chain}{number:4d}    "
        f"{xyz[0]:8.3f}{xyz[1]:8.3f}{xyz[2]:8.3f}  1.00 80.00          {element:>2s}  "
    )

def test_split_and_connectivity(canonical: str) -> None:
    atoms = split_atoms(canonical)
    validate_assembly(atoms)
    assert len([a for a in atoms if a.name == "CA"]) == 1312
    assert any(a.chain == "E" and a.residue == 1 and a.name == "OG1" for a in atoms)
    text = design_pdb(atoms, canonical)
    assert text.count("TER\n") == 9
    assert "CONECT 1316 1317 1318\n" in text
    assert " H1 " not in text
    assert {a.serial: a.xyz for a in atoms} == {
        int(line[6:11]): tuple(float(line[start : start + 8]) for start in (30, 38, 46))
        for line in text.splitlines()
        if line[:6] in {"ATOM  ", "HETATM"}
    }


@pytest.mark.parametrize(
    "kind", ["empty", "element", "duplicate", "residue", "alternate", "nan"]
)
def test_parser_rejects_invalid_atoms(canonical: str, kind: str) -> None:
    first = canonical.splitlines()[0]
    text = {
        "empty": "END\n",
        "element": first[:76] + "  ",
        "duplicate": first + "\n" + first,
        "residue": first[:21] + "Z" + first[22:],
        "alternate": first[:16] + "A" + first[17:],
        "nan": first[:30] + "     nan" + first[38:],
    }[kind]
    with pytest.raises(ValueError):
        split_atoms(text)


def test_assembly_rejects_missing_or_changed_residues(canonical: str) -> None:
    atoms = split_atoms(canonical)
    with pytest.raises(ValueError, match="complete"):
        validate_assembly(atoms[1:])
    for original, replacement, message in (
        ("PHE", "ALA", "Oxyanion"),
        ("THR", "ALA", "Catalytic"),
    ):
        with pytest.raises(ValueError, match=message):
            validate_assembly(split_atoms(canonical.replace(original, replacement)))
    changed = canonical.replace("ALA B   1", "GLY B   1")
    with pytest.raises(ValueError, match="sequences differ"):
        validate_assembly(split_atoms(changed))


def test_mask_symmetry_and_commands(canonical: str) -> None:
    atoms = split_atoms(canonical)
    alpha, beta = focused_positions([atoms], ["C54", "O49"], 8, [113, 186], [1, 2, 3])
    assert alpha == [53]
    assert beta == [5]
    redesigned, ties, weights = symmetry_tokens(alpha, beta)
    assert redesigned == "A53 B53 C53 D53 E5 F5 G5 H5"
    assert ties == "A53,B53,C53,D53|E5,F5,G5,H5"
    assert weights == "0.25,0.25,0.25,0.25|0.25,0.25,0.25,0.25"
    command = mpnn_command(
        {
            "ligandmpnn_python": "/env/python",
            "ligandmpnn_directory": "/mpnn",
            "ligandmpnn_checkpoint": "/weights.pt",
            "sequences_per_temperature_per_backbone": 3,
        },
        Path("/in.pdb"),
        Path("/output"),
        alpha,
        beta,
        0.1,
        123,
    )
    assert command[:2] == ["/env/python", "/mpnn/run.py"]
    assert command[command.index("--homo_oligomer") + 1] == "0"
    assert command[command.index("--symmetry_residues") + 1] == ties
    assert command[command.index("--ligand_mpnn_use_side_chain_context") + 1] == "1"

            list(range(1, 97)),
        )


def test_alignment_measures_deformation_not_rigid_motion(canonical: str) -> None:
    from dataclasses import replace

    atoms = split_atoms(canonical)
    rotation = np.array([[0, -1, 0], [1, 0, 0], [0, 0, 1]])
    moved = [replace(a, xyz=tuple(np.array(a.xyz) @ rotation + 10)) for a in atoms]
    assert aligned_rmsd(atoms, moved) < 1e-10
    changed = [
        replace(a, xyz=(a.xyz[0] + 1, a.xyz[1], a.xyz[2]))
        if a.chain == "A" and a.residue == 53
        else a
        for a in moved
    ]
    assert aligned_rmsd(atoms, changed) > 0.1
    with pytest.raises(ValueError, match="Incomplete"):
        aligned_rmsd(atoms, atoms[1:])

def test_prepare_freezes_inputs_and_refuses_overwrite(
    preparation: tuple[Path, Path],
) -> None:
    config_path, output = preparation
    manifest = runner.prepare(config_path, output)
    assert manifest["status"] == "prepared_not_launched"
    assert len(manifest["jobs"]) == 4
    assert manifest["expected_sequences_before_deduplication"] == 12
    assert manifest["design_mask"]["alpha_positions"] == [53]
    assert json.loads((output / "preparation_manifest.json").read_text()) == manifest
    with pytest.raises(FileExistsError):
        runner.prepare(config_path, output)


@pytest.mark.parametrize(
    "field,value",
    [
        ("schema_version", 2),
        ("reacting_atoms", ["C12", "O4"]),
        ("protected_alpha_positions", [113]),
        ("protected_beta_positions", []),
        ("models", [40, 40]),
        ("temperatures", [0]),
        ("sequences_per_temperature_per_backbone", 0),
    ],
)
def test_prepare_rejects_incompatible_config(
    preparation: tuple[Path, Path], field: str, value: object
) -> None:
    config_path, output = preparation
    config = json.loads(config_path.read_text())
    config[field] = value
    config_path.write_text(json.dumps(config))
    with pytest.raises(ValueError):
        runner.prepare(config_path, output)
    assert not output.exists()


def test_prepare_missing_model_fails_before_output(
    preparation: tuple[Path, Path],
) -> None:
    config_path, output = preparation

    config["models"] = [40, 99]
    config_path.write_text(json.dumps(config))
    with pytest.raises(ValueError, match="Tier A/B"):
        runner.prepare(config_path, output)
