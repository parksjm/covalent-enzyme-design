"""Measure rigid transforms between alpha/beta repeats without assigning symmetry."""

from __future__ import annotations

import hashlib
import json
import logging
from pathlib import Path

import numpy as np

ROOT = Path("/data/jpc/covalent-enzyme-design")
SOURCE = ROOT / "output_laptop_inspection_20261021/protein/pa6_l4_ti1_starting.pdb"
OUTPUT = ROOT / "output_nyl12_parameter_validation_20261022/repeat_symmetry_audit_01"
LOGGER = logging.getLogger(__name__)


def main() -> None:
    """Record CA correspondence, Kabsch transforms and rotation-axis comparisons."""
    logging.basicConfig(level=logging.INFO)
    atoms: dict[tuple[str, int], np.ndarray] = {}
    identities: dict[tuple[str, int], str] = {}
    for line in SOURCE.read_text().splitlines():
        if not line.startswith("ATOM  ") or line[12:16].strip() != "CA":
            continue
        key = (line[21], int(line[22:26]))
        if key in atoms:
            raise ValueError(f"Duplicate CA: {key}")
        atoms[key] = np.array(
            [float(line[30:38]), float(line[38:46]), float(line[46:54])]
        )
        identities[key] = line[17:20]
    repeats = []
    for alpha, beta in zip("ABCD", "EFGH", strict=True):
        keys = [(alpha, i) for i in range(1, 233)] + [(beta, i) for i in range(1, 97)]
        if set(k for k in atoms if k[0] in (alpha, beta)) != set(keys):
            raise ValueError("Unexpected WT chain lengths or residue numbers")
        if repeats:
            reference_keys = [("A", i) for i in range(1, 233)] + [
                ("E", i) for i in range(1, 97)
            ]
            if any(
                identities[a] != identities[b]
                for a, b in zip(reference_keys, keys, strict=True)
            ):
                raise ValueError(
                    "Repeat sequences differ; resolve correspondence explicitly"
                )
        repeats.append(np.array([atoms[k] for k in keys]))
    reference = repeats[0]
    center = reference.mean(axis=0)
    records = []
    axes = []
    for pair, points in zip(("BF", "CG", "DH"), repeats[1:], strict=True):
        u, _, vt = np.linalg.svd(
            (reference - center).T @ (points - points.mean(axis=0))
        )
        correction = np.eye(3)
        correction[-1, -1] = np.linalg.det(u @ vt)
        rotation = (u @ correction @ vt).T
        translation = points.mean(axis=0) - rotation @ center
        predicted = reference @ rotation.T + translation
        angle = float(
            np.rad2deg(np.arccos(np.clip((np.trace(rotation) - 1) / 2, -1, 1)))
        )
        _, _, axis_vectors = np.linalg.svd(rotation - np.eye(3))
        axis = axis_vectors[-1]
        axes.append(axis)
        records.append(
            {
                "from": "AE",
                "to": pair,
                "ca_count": len(reference),
                "rmsd_A": float(
                    np.sqrt(np.mean(np.sum((predicted - points) ** 2, axis=1)))
                ),
                "rotation_degrees": angle,
                "rotation": rotation.tolist(),
                "translation_A": translation.tolist(),
                "axis_unsigned": axis.tolist(),
            }
        )
    snapshot = OUTPUT / "source"
    snapshot.mkdir()
    payload = SOURCE.read_bytes()
    (snapshot / SOURCE.name).write_bytes(payload)
    (snapshot / Path(__file__).name).write_bytes(Path(__file__).read_bytes())
    report = {
        "scope": "Geometric diagnostic on retained AF3-derived WT scaffold; not a crystal biological-assembly assignment",
        "point_group_assigned": False,
        "repeat_pairs": ["AE", "BF", "CG", "DH"],
        "protein_source": str(SOURCE),
        "protein_sha256": hashlib.sha256(payload).hexdigest(),
        "script_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "reference_to_copy_transforms": records,
        "axis_absolute_dot_products": np.abs(
            np.array(axes) @ np.array(axes).T
        ).tolist(),
        "protein_refinement_or_design_performed": False,
    }
    (OUTPUT / "report.json").write_text(json.dumps(report, indent=2) + "\n")
    LOGGER.info("Repeat rotation angles: %s", [r["rotation_degrees"] for r in records])
    LOGGER.info("Repeat CA RMSD: %s", [r["rmsd_A"] for r in records])


if __name__ == "__main__":
    main()
