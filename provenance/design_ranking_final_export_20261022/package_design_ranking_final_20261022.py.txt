"""Package design-ranking documentation with verified source and payload digests."""

from __future__ import annotations

import fcntl
import hashlib
import json
import logging
import re
import shutil
import sys
from pathlib import Path

ROOT = Path("/data/jpc/covalent-enzyme-design")
sys.path.insert(0, str(ROOT))
from scripts.prepare_nyl12_inspection_download import build_package  # noqa: E402

LOGGER = logging.getLogger(__name__)
WORK = ROOT / "output_nyl12_parameter_validation_20261022"
ARCHIVE = ROOT / "provenance/design_ranking_final_export_20261022"


def digest(path: Path) -> str:
    """Return a small file's SHA-256 identity."""
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main() -> None:
    """Verify prerequisites, freeze evidence, build and promote a fresh package."""
    logging.basicConfig(level=logging.INFO)
    status = json.loads(
        (WORK / "ranking_export_verification_02/job/status.json").read_text()
    )
    if status["exit_status"] != 0 or not status["ended_utc"]:
        raise ValueError("Export verification must complete successfully")
    for name, expected in status["input_sha256"].items():
        if digest(Path(name)) != expected:
            raise ValueError(f"Verified source changed: {name}")
    methods = ROOT / "manuscript/METHODS_DRAFT.md"
    references = ROOT / "manuscript/REFERENCES.md"
    used = set(re.findall(r"\bR\d+\b", methods.read_text()))
    defined = re.findall(r"\*\*(R\d+)\.", references.read_text())
    if len(set(defined)) != len(defined) or not used.issubset(defined):
        raise ValueError("Missing or duplicate manuscript reference identifiers")
    current = ROOT / "output_laptop_inspection_20261021"
    staging = ROOT / "output_laptop_inspection_20261021_ranking_final_staging"
    backup = ROOT / "output_laptop_inspection_20261021_revision_14"
    if staging.exists() or backup.exists():
        raise FileExistsError("Fresh package staging and backup required")
    ARCHIVE.mkdir()
    source_records = {}
    source_paths = [
        methods,
        ROOT / "docs/design_ranking_and_iteration_20261022.md",
        ROOT / "docs/standalone_handoff_20261020.md",
        references,
        ROOT / "manuscript/SOURCE_LEDGER.md",
        ROOT / "scripts/pull_nyl12_inspection.sh",
        ROOT / "scripts/prepare_nyl12_inspection_download.py",
        Path(__file__),
        Path("/data/jpc/code/ambertools26/venv/dat/leap/parm/gaff2.dat"),
    ]
    local_source = WORK / "ranking_package_02/source"
    local_source.mkdir(parents=True)
    for source in source_paths:
        target = local_source / source.name
        shutil.copy2(source, target)
        if digest(target) != digest(source):
            raise ValueError(f"Source changed while copying: {source}")
        source_records[str(source)] = {
            "sha256": digest(source),
            "snapshot": str(target),
        }
    frozen_journal = ARCHIVE / "agent_actions_snapshot.jsonl"
    with (ROOT / "output_agent_actions/20261021.jsonl").open("rb") as source:
        fcntl.flock(source, fcntl.LOCK_SH)
        with frozen_journal.open("xb") as target:
            shutil.copyfileobj(source, target)
        fcntl.flock(source, fcntl.LOCK_UN)
    old = json.loads((current / "provenance.json").read_text())
    files = {name: Path(row["source"]) for name, row in old["files"].items()}
    for directory in ("docs", "manuscript"):
        for source in sorted((ROOT / directory).glob("*.md")):
            files[f"{directory}/{source.name}"] = source
    files["README.md"] = ROOT / "docs/laptop_inspection_20261021.md"
    files["provenance/agent_actions_at_export.jsonl"] = frozen_journal
    files["reports/redesign/repeat_symmetry.json"] = (
        WORK / "repeat_symmetry_audit_01/report.json"
    )
    files["reports/parameters/gaff2_typing_inventory.json"] = (
        WORK / "gaff2_typing_inventory_02/report.json"
    )
    files["reports/parameters/gaff2_junction_review.json"] = (
        WORK / "gaff2_junction_review_01/report.json"
    )
    files["reports/redesign/recent_protected_preparation.json"] = (
        ROOT
        / "provenance/redesign_symmetry_and_gaff2_20261022/recent_protected_preparation_summary.json"
    )
    result = build_package(files, staging)
    for line in (staging / "MANIFEST.sha256").read_text().splitlines():
        expected, name = line.split("  ", 1)
        if digest(staging / name) != expected:
            raise ValueError(f"Package digest mismatch: {name}")
    required = (
        "docs/design_ranking_and_iteration_20261022.md",
        "docs/standalone_handoff_20261020.md",
        "manuscript/METHODS_DRAFT.md",
        "manuscript/REFERENCES.md",
        "manuscript/SOURCE_LEDGER.md",
        "manuscript/README.md",
    )
    for name in required:
        if (staging / name).read_bytes() != (ROOT / name).read_bytes():
            raise ValueError(f"Current manuscript missing from package: {name}")
    for name in old["files"]:
        if name.startswith(("protein/", "qm/", "af3/", "optimization/")):
            if (current / name).read_bytes() != (staging / name).read_bytes():
                raise ValueError(f"Existing structural payload changed: {name}")
    report = {
        "verified": True,
        "reference_ids_used": sorted(used),
        "reference_ids_defined": defined,
        "required_manuscript_files": {
            name: digest(staging / name) for name in required
        },
        "payload_files": len(result["files"]),
        "payload_bytes": result["payload_bytes"],
        "retained_backup": str(backup),
        "source_records": source_records,
        "structure_and_trajectory_payload_unchanged": True,
    }
    current.rename(backup)
    staging.rename(current)
    (ARCHIVE / "package_verification.json").write_text(
        json.dumps(report, indent=2) + "\n"
    )
    LOGGER.info(
        "Verified Methods/References package: %d files, %d bytes",
        report["payload_files"],
        report["payload_bytes"],
    )


if __name__ == "__main__":
    main()
