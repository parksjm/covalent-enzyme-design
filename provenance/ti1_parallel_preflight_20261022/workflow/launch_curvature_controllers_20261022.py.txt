"""Freeze verified analysis sources and launch four post-frequency waiters."""
from __future__ import annotations

import hashlib
import json
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path('/data/jpc/covalent-enzyme-design')
BASE = ROOT / 'output_nyl12_ti1_parameters_20261022/pa6_local_bonded'
PYTHON = ROOT / '.venv/bin/python'
SNAPSHOT = BASE / 'conditional_curvature_source'


def digest(path: Path) -> str:
    """Return source SHA-256."""
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main() -> None:
    """Require verified sources, then retain a repo-shaped analysis snapshot."""
    validation = ROOT / 'output_nyl12_parameter_validation_20261022'
    for job in ('parallel_full_verification_02', 'curvature_waiter_verification_01'):
        status = json.loads((validation / job / 'job/status.json').read_text())
        if status['exit_status'] != 0 or not status['ended_utc']:
            raise ValueError(f'Verification not complete: {job}')
        for name, expected in status['input_sha256'].items():
            if digest(Path(name)) != expected:
                raise ValueError(f'Verified source changed: {name}')
    names = ['scripts/analyze_nyl12_ti1_curvature.py',
             'scripts/wait_nyl12_ti1_curvature.py', 'scripts/record_agent_action.py',
             'pyproject.toml', 'uv.lock']
    names += [f'src/nyl12_covalent/{name}.py' for name in (
        '__init__', 'constrained_curvature', 'dihedral_constraints',
        'gaussian_fchk', 'gaussian_checks', 'ti1_reference_coordinates')]
    SNAPSHOT.mkdir()
    for name in names:
        target = SNAPSHOT / name
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(ROOT / name, target)
    identity = {name: digest(SNAPSHOT / name) for name in names}
    (SNAPSHOT / 'source_identity.json').write_text(json.dumps(identity, indent=2)+'\n')
    for case in sorted(BASE.glob('attack_*')):
        if not (case / 'frequency_controller_01/job/status.json').is_file():
            raise FileNotFoundError(f'Missing existing frequency controller: {case}')
        subprocess.run([
            str(PYTHON), str(ROOT / 'scripts/run_nyl12_job.py'), 'launch',
            '--session', f'nyl12_ti1_curvature_{case.name}_20261022',
            '--output-dir', str(case / 'conditional_curvature_controller_01/job'),
            '--cwd', str(ROOT), '--env', 'OPENBLAS_NUM_THREADS=1',
            '--input', str(SNAPSHOT / 'source_identity.json'),
            '--input', str(SNAPSHOT / 'scripts/wait_nyl12_ti1_curvature.py'),
            '--', str(PYTHON), str(SNAPSHOT / 'scripts/wait_nyl12_ti1_curvature.py'),
            '--case', str(case), '--snapshot', str(SNAPSHOT),
            '--journal', str(ROOT / 'output_agent_actions/20261021.jsonl'),
            '--wait-hours', '48', '--poll-seconds', '60',
        ], check=True)


if __name__ == '__main__':
    main()
