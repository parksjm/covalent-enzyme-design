"""Audit installed MCPB bond units using manufactured quadratic Hessians."""
from __future__ import annotations

import hashlib
import inspect
import json
import logging
from pathlib import Path

import numpy as np
from pymsmt.mcpb import gene_final_frcmod_file as fitting
from pymsmt.mol import constants

OUT = Path('/data/jpc/covalent-enzyme-design/output_nyl12_parameter_validation_20261022/mcpb_bond_conventions_02')
LOGGER = logging.getLogger(__name__)


def main() -> None:
    """Verify isolated-bond conventions without using any molecular QM result."""
    logging.basicConfig(level=logging.INFO)
    rows = []
    for direction in (np.array([1., 0., 0.]), np.array([1., 2., 3.])/np.sqrt(14.)):
        coords = np.concatenate((np.zeros(3), 2.3*direction))
        derivative = np.concatenate((-direction,direction))
        curvature = 0.125
        hessian = curvature*np.outer(derivative,derivative)
        for scaling in (1.,0.95):
            for average in (0,1):
                observed = fitting.get_bond_fc_with_sem([],coords,hessian,1,2,scaling,average)
                expected_r = 2.3*constants.B_TO_A
                expected_k = curvature*constants.HB2_TO_KCAL_MOL_A2*0.5*scaling**2
                if not np.isclose(observed[0],expected_r,rtol=1e-12,atol=1e-12):
                    raise ValueError('Bond length unit mismatch')
                if not np.isclose(observed[1],expected_k,rtol=1e-10,atol=1e-10):
                    raise ValueError('AMBER force constant convention mismatch')
                observed_k = complex(observed[1])
                if abs(observed_k.imag) > 1e-12*max(1.0,abs(observed_k.real)):
                    raise ValueError('Non-negligible imaginary force constant')
                rows.append({'direction':direction.tolist(),'frequency_scaling':scaling,
                             'average_blocks':average,'observed_r_A':float(observed[0]),
                             'observed_amber_k':observed_k.real,
                             'observed_amber_k_imaginary_residual':observed_k.imag,
                             'expected_amber_k':expected_k,'accepted':True})
    sources = [Path(inspect.getfile(fitting)),Path(inspect.getfile(constants)),Path(__file__)]
    report = {'accepted':True,'scope':'Analytic isolated-bond units/conventions only',
              'forcefield_validated':False,'ti1_parameters_fitted':False,
              'modified_angle_validation_pending':True,'cases':rows,
              'input_conventions':{'coordinates':'bohr','hessian':'Hartree/bohr^2',
                                   'atom_indices':'one-based','harmonic_energy':'K*(r-r0)^2'},
              'source_sha256':{str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in sources}}
    (OUT/'report.json').write_text(json.dumps(report,indent=2)+'\n')
    LOGGER.info('Accepted %d manufactured bond checks; no TI1 parameters fitted',len(rows))


if __name__ == '__main__':
    main()
